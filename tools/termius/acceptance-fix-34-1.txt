# VWARD acceptance fix.34.1 - only reads, changes nothing.
# Standard Termius v2.3.3. 4 pastes, each under 0.8 KB, one by one.
# Paste 1-3 write /tmp/acc341.sh, paste 4 runs it.

# ---- paste 1 ----
cat > /tmp/acc341.sh <<'EOF'
clear
OUT="/tmp/vward-acc341-$(date +%Y%m%d-%H%M%S).txt"
LAST="/tmp/last_result.txt"
ESC=$(printf '\033')
: > "$OUT"; ln -sfn "$OUT" "$LAST"
send_out() {
  [ -s "$OUT" ] || return 1
  sed "s/${ESC}\[[0-9;?]*[a-zA-Z]//g" "$OUT" |
    tail -c 30000 | base64 | tr -d '\r\n' |
    { read -r B; printf '\033]52;c;%s\a' "$B"; }
}
trap 'trap "" INT TERM HUP; send_out; exit 130' INT TERM HUP
ok() { echo "[ PASS ] $*"; }
bad() { echo "[ FAIL ] $*"; }
S=/opt/var/lib/vward/policy-sync
C=/opt/etc/vward/route-engine/domain-lists.conf
PL=/opt/var/log/vward-policy-sync-sync.log
EOF

# ---- paste 2 ----
cat >> /tmp/acc341.sh <<'EOF'
{
echo "=== VWARD ACCEPTANCE fix.34.1 ==="
echo "Date: $(date '+%F %T') up: $(cut -d' ' -f1 /proc/uptime)s"
echo "Version: $(cat /opt/share/vward/VERSION 2>/dev/null)"
echo "guard: $(grep -E '^(smartdns_guard|shrink_guard)=' $C \
  2>&1 | tr '\n' ' ')"
echo "=== 1. IP CATEGORIES ==="
A=$(grep -c . $S/active.categories 2>/dev/null)
[ "${A:-0}" -ge 9 ] && ok "active=$A" || bad "active=${A:-none}"
H=$(grep -c . $S/smartdns.categories 2>/dev/null)
[ "${H:--}" = 0 ] && ok "held empty" || bad "held=${H:-no file}"
echo "blocks=$(grep -c . $S/smartdns.addresses 2>/dev/null)"
L=$(grep ' SYNC ' $PL 2>/dev/null | tail -n 1)
echo "${L:-no SYNC line}" | cut -c1-150
case "$L" in *' removed=0 '*) ok "removed=0" ;;
  *) bad "removed not 0 or no line" ;; esac
EOF

# ---- paste 3 ----
cat >> /tmp/acc341.sh <<'EOF'
echo "=== 2. SHRINK GUARD, LAST GOOD ==="
if [ -f $S/shrink.pending ]; then
  echo "[ WARNING ] shrink.pending:"; cat $S/shrink.pending
else echo "shrink.pending: none"; fi
echo "owned=$(grep -c . $S/owned.dynamic.routes 2>/dev/null)" \
  "last-good=$(grep -c . $S/owned.last-good 2>/dev/null)"
grep -E 'SHRINK|SKIP reconcile|RESTORE' $PL 2>/dev/null |
  tail -n 3 | cut -c1-150
echo "=== 3. DNS RULES (r6) ==="
R=/tmp/vward-dns-rules
[ -f $R ] && tr '\n' ' ' < $R && echo || echo "no state yet"
grep -q '^RESULT=ok' $R 2>/dev/null && grep -q '^LIVE=16' $R &&
  grep -q '^CONFIG=16' $R && ok "16 of 16" || bad "not 16 of 16"
grep 'DNS_RULES' /opt/var/log/vward-tunnel-health.log 2>/dev/null |
  tail -n 2 | cut -c1-150
EOF

# ---- paste 4 ----
cat >> /tmp/acc341.sh <<'EOF'
for F in /var/ndnproxymain.conf /tmp/ndnproxymain.conf; do
  [ -r $F ] && echo "$F: $(grep -ci youtube $F) yt lines" &&
    grep -i 'youtube.com' $F | head -n 2 | cut -c1-80
done
echo "=== END ==="
} 2>&1 | tee "$OUT"
P=$(grep -c '^\[ PASS \]' "$OUT"); F=$(grep -c '^\[ FAIL \]' "$OUT")
echo "SUMMARY: pass=$P fail=$F" | tee -a "$OUT"
if send_out; then echo "RESULT SENT (OSC52)"
else echo "[ FAIL ] nothing to send"; fi
echo "RAW: $LAST"
EOF
sh -n /tmp/acc341.sh && sh /tmp/acc341.sh
