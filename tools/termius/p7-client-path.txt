cat > /tmp/p7-client-path.sh <<'EOF'
clear
umask 077
PFX=vward-p7-path
OUT="/tmp/$PFX-$(date +%Y%m%d-%H%M%S).txt"
MAXB=30000
ESC=$(printf '\033')
ls -1t /tmp/$PFX-*.txt 2>/dev/null |
  tail -n +5 |
  while read -r F; do rm -f "$F"; done
printf '' > "$OUT" || exit 1
ln -sfn "$OUT" /tmp/last_result.txt
ln -sfn "$OUT" "/tmp/last_$PFX.txt"
TMP=$(mktemp -d /tmp/vward-p7.XXXXXX) || exit 1
tmo() {
  _ts=$1; shift
  "$@" & _tp=$!
  ( sleep "$_ts"; kill "$_tp" 2>/dev/null ) & _tw=$!
  wait "$_tp"; _tr=$?
  kill "$_tw" 2>/dev/null
  return $_tr
}
send_out() {
  sed "s/${ESC}\[[0-9;?]*[a-zA-Z]//g" "$OUT" |
    tail -c "$MAXB" | base64 | tr -d '\r\n' |
    { read -r B; printf '\033]52;c;%s\a' "$B"; }
}
on_abort() {
  trap '' INT TERM HUP
  send_out
  rm -rf "$TMP"
  echo '[ ABORTED / DISCONNECTED ]:'
  echo 'PARTIAL RESULT SAVED TO /tmp/last_result.txt'
  exit 130
}
trap on_abort INT TERM HUP
diag() {
  echo 'P7 READ ONLY: no configuration changes'
  date '+%Y-%m-%d %H:%M:%S %z'
  tmo 5 ndmc -c 'show version' > "$TMP/ver" 2>&1
  VRC=$?
  grep -E 'release|title|device' "$TMP/ver" |
    head -n 3
  echo "VERSION_RC=$VRC"
  [ "$VRC" = 0 ] || echo '[ WARNING ] no version reply'
  MEM=$(awk '/MemFree:|Buffers:|^Cached:/ {s+=$2}
    /MemAvailable:/ {a=$2}
    END {print a ? a : s}' /proc/meminfo)
  echo "[ OK ] MEMORY_KB=$MEM"
EOF

    *[!0-9a-fA-F:]*|'') MAC= ;;
  esac
  [ "${#MAC}" = 17 ] || MAC=
  echo "[ OK ] LAN_MAC=${MAC:-unknown}"

  echo '[ WARNING ] client DNS/DoH is not inferred'
  echo '[ OK ] ROUTER DNS A/AAAA (not client replies)'
  for Q in A AAAA; do
    tmo 5 nslookup -type="$Q" "$H" \
EOF

    RC=$?

  done
  echo '[ OK ] CLIENT ROUTING POLICY'
  tmo 3 ip rule show | head -n 15

  echo '[ OK ] OPEN ONE APP NOW: CAPTURE 15s'
  command -v tcpdump >/dev/null 2>&1 || {
    echo '[ NOT_FOUND ] tcpdump'; return
  }
  # LAN identifies the client; WAN/tunnel destination
  # headers establish correlation across NAT. No payload.
  PIDS=
  for D in "$VWARD_LAN_DEVICE" "$VWARD_WAN_DEVICE" \
    "$VWARD_TUNNEL_DEVICE"; do
    case "$D" in
      ''|*[!A-Za-z0-9_.:-]*) continue ;;
    esac
    FILTER='udp or tcp port 443 or tcp port 53'
    if [ "$D" = "$VWARD_LAN_DEVICE" ]; then
      if [ -n "$MAC" ]; then
        FILTER="ether host $MAC and ($FILTER)"
      else
        FILTER="host $CLIENT and ($FILTER)"
      fi
    fi
    tmo 15 tcpdump -p -nni "$D" -q -c 50 \
      "$FILTER" > "$TMP/cap-$D" 2>&1 &
    PIDS="$PIDS $!"
  done
  for P in $PIDS; do wait "$P"; done
  for F in "$TMP"/cap-*; do
    [ -f "$F" ] || continue
    echo "[ OK ] ${F##*/} IPv4/IPv6 TCP/UDP HEADERS"
    head -n 54 "$F"
  done
  echo '[ WARNING ] Match LAN destinations to WAN/tunnel.'
  echo 'No matching packets is insufficient evidence.'
EOF
