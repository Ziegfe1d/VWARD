cat > /tmp/vward-p7-path.sh <<'EOF'
clear
umask 077
PFX=vward-p7-path
OUT="/tmp/$PFX-$(date +%Y%m%d-%H%M%S).txt"
MAXB=30000
ESC=$(printf '\033')
ls -1t /tmp/$PFX-*.txt 2>/dev/null |
  tail -n +5 |
  while read -r F; do rm -f "$F"; done
printf '' > "$OUT" || exit 1
ln -sfn "$OUT" /tmp/last_result.txt
ln -sfn "$OUT" "/tmp/last_$PFX.txt"
TMP=$(mktemp -d /tmp/vward-p7.XXXXXX) || exit 1
tmo() {
  _ts=$1; shift
  "$@" & _tp=$!
  ( sleep "$_ts"; kill "$_tp" 2>/dev/null ) & _tw=$!
  wait "$_tp"; _tr=$?
  kill "$_tw" 2>/dev/null
  return $_tr
}
send_out() {
  sed "s/${ESC}\[[0-9;?]*[a-zA-Z]//g" "$OUT" |
    tail -c "$MAXB" | base64 | tr -d '\r\n' |
    { read -r B; printf '\033]52;c;%s\a' "$B"; }
}
on_abort() {
  trap '' INT TERM HUP
  send_out
  rm -rf "$TMP"
  echo '[ ABORTED ] RAW: /tmp/last_result.txt'
  exit 130
}
trap on_abort INT TERM HUP
diag() {
  echo 'P7 READ ONLY: no configuration changes'
  date '+%Y-%m-%d %H:%M:%S %z'
  echo 'Open ONE affected app during the capture.'
  L=/opt/lib/vward/vward-device-profile.sh
  [ -r "$L" ] || { echo 'FAIL: no profile'; return; }
  . "$L"
  vward_profile_load || { echo 'FAIL: profile'; return; }
  R=/opt/share/vward/updater/current
  jq -r '.platform_version // "unknown"' \
    "$R/component-registry.json" 2>/dev/null
EOF

cat >> /tmp/vward-p7-path.sh <<'EOF'
  tmo 12 ndmc -c 'show running-config' > "$TMP/rc"
  echo "CONFIG_RC=$?"
  awk '/^object-group fqdn / {g=$3} /^!/ {g=""}
    g=="domain-list11" && $1=="include" {print $2}' \
    "$TMP/rc" | head -n 2 > "$TMP/games"
  printf '%s\n' api.ozon.ru lampa.mx > "$TMP/hosts"
  cat "$TMP/games" >> "$TMP/hosts"
  echo '=== RECENT OZON DECISIONS ==='
  E=/opt/var/log/vward-route-engine-events.log
  tail -n 500 "$E" 2>/dev/null |
    grep -Fi 'api.ozon.ru' | tail -n 4
  echo '=== DNS SOURCES AND ROUTER IPv4 ROUTES ==='
  while read -r H; do
    for S in "$VWARD_DNS_SERVER" "$VWARD_PROBE_DNS"; do
      [ -n "$S" ] || continue
      echo "DOMAIN=$H DNS=$S"
      tmo 5 nslookup "$H" "$S" > "$TMP/dns" 2>&1
      echo "DNS_RC=$?"
      awk '/^Name:/ {a=1} a && /^Address/ {print}' \
        "$TMP/dns" | head -n 4
      awk '/^Name:/ {a=1}
        a && /^Address/ && $NF ~ /^[0-9.]+$/ \
        {print $NF}' "$TMP/dns" |
        head -n 2 > "$TMP/ips"
      while read -r I; do
        tmo 3 ip route get "$I" 2>&1 | head -n 1
      done < "$TMP/ips"
    done
  done < "$TMP/hosts"
  echo '=== CLIENT / WAN / TUNNEL HEADERS: 12s ==='
  echo 'No packet payload is captured.'
  if ! command -v tcpdump >/dev/null 2>&1; then
EOF

cat >> /tmp/vward-p7-path.sh <<'EOF'
    echo 'SKIPPED: tcpdump unavailable'
    return
  fi
  PIDS=
  for D in "$VWARD_LAN_DEVICE" "$VWARD_WAN_DEVICE" \
    "$VWARD_TUNNEL_DEVICE"; do
    case "$D" in ''|*[!A-Za-z0-9_.:-]*) continue ;; esac
    tmo 12 tcpdump -nni "$D" -q -c 16 \
      'udp or tcp port 443' > "$TMP/cap-$D" 2>&1 &
    PIDS="$PIDS $!"
  done
  for P in $PIDS; do wait "$P"; done
  for F in "$TMP"/cap-*; do
    [ -f "$F" ] || continue
    echo "--- ${F##*/} ---"
    head -n 20 "$F"
  done
  echo 'SUMMARY: client path requires matching headers.'
  echo 'Router ip-route-get alone is not client proof.'
}
diag 2>&1 | tee "$OUT"
rm -rf "$TMP"
N=$(wc -c < "$OUT")
[ "$N" -le "$MAXB" ] || echo "WARNING: size=$N"
send_out
echo
echo 'RESULT SENT (OSC52)'
echo 'RAW: /tmp/last_result.txt'
EOF

/opt/bin/sh -n /tmp/vward-p7-path.sh && \
  /opt/bin/sh /tmp/vward-p7-path.sh
