Diagnostic (11.11.2026): why new SSH sessions die before the login. Pastes 1 and 2
change nothing on the router: they download the Entware packages that own libc,
libm, libnss_files, libcrypt into /tmp (plain HTTP), compare SHA-256 of the installed
files with them and remove the download. No secrets are printed. Paste 3 is optional.
Needs a working shell; without one the Panel shows «Вход по SSH» and the journal
line SSH_LOGIN_BROKEN|user_lookup_rc=... after the next update.

=== PASTE 1 ===
cat > /tmp/dg12.sh <<'EOF'
clear
PFX="vward-diag12"
OUT="/tmp/${PFX}-$(date +%Y%m%d-%H%M%S).txt"
LAST="/tmp/last_result.txt"
MAXB=30000
KEEP=5
ESC=$(printf '\033')
ls -1t /tmp/${PFX}-*.txt 2>/dev/null |
  tail -n +"$KEEP" |
  while read -r F; do rm -f "$F"; done
: > "$OUT"
ln -sfn "$OUT" "$LAST"
ln -sfn "$OUT" "/tmp/last_${PFX}.txt"
send_out() {
  [ -s "$OUT" ] || return 1
  sed "s/${ESC}\[[0-9;?]*[a-zA-Z]//g" "$OUT" |
    tail -c "$MAXB" | base64 | tr -d '\r\n' |
    { read -r B; printf '\033]52;c;%s\a' "$B"; }
}
on_abort() {
  trap '' INT TERM HUP
  send_out
  printf '\n[ ABORTED ]: PARTIAL IN %s\n' "$LAST"
  exit 130
}
trap on_abort INT TERM HUP
EOF

=== PASTE 2 ===
cat >> /tmp/dg12.sh <<'EOF'
W=/tmp/dg12-pkg; rm -rf $W; mkdir -p $W
{
echo " VWARD DIAG-12 SSH/libs $(date '+%Y-%m-%d %H:%M:%S') ver: $(cat /opt/share/vward/VERSION 2>/dev/null)"
echo "=== 1. USER LOOKUP IN A FRESH PROCESS (libnss_files; 132=SIGILL 139=SIGSEGV) ==="
/opt/bin/busybox id root >/dev/null 2>&1; echo "id root rc=$?"
/opt/bin/busybox id nobody >/dev/null 2>&1; echo "id nobody rc=$?"
grep -v '^#' /opt/etc/nsswitch.conf 2>/dev/null | grep -E 'passwd|group' | head -n 2
echo "=== 2. LIBS OF DROPBEAR (ldd) ==="
LD_TRACE_LOADED_OBJECTS=1 /opt/sbin/dropbear 2>&1 | head -n 12
echo "=== 3. SHA OF THE FILES vs THEIR ENTWARE PACKAGE ==="
F=/opt/etc/opkg.conf; FEED=$(awk '/^src/ {print $3; exit}' $F)
echo "feed: $FEED"
for L in libnss_files libc.so libm.so libcrypt ld-linux; do
  grep -l "/$L" /opt/lib/opkg/info/*.list 2>/dev/null
done | sort -u | while read -r LST; do
  P=$(basename "$LST" .list)
  V=$(opkg status "$P" 2>/dev/null | awk '/^Version:/ {print $2}')
  FN=$(awk -v p="$P" '$1=="Package:" {f=($2==p)} f && $1=="Filename:" {print $2; exit}' /opt/var/opkg-lists/* 2>/dev/null)
  echo "-- package $P $V file=${FN:-?}"
  [ -n "$FN" ] || continue
  mkdir -p $W/$P/d
  /opt/bin/curl -s -m 60 -o $W/$P/p.ipk "$FEED/$FN" || wget -q -O $W/$P/p.ipk "$FEED/$FN"
  ( cd $W/$P && tar -xzf p.ipk 2>/dev/null && tar -xzf data.tar.gz -C d 2>/dev/null ) || { echo "   cannot unpack"; continue; }
  grep -E '/(libnss_files|libc[.-]|libm[.-]|libcrypt|ld-)' "$LST" | while read -r FILE; do
    R=$W/$P/d$FILE
    [ -f "$R" ] || continue
    A=$(sha256sum "$FILE" 2>/dev/null | cut -c1-16); B=$(sha256sum "$R" | cut -c1-16)
    [ "$A" = "$B" ] && echo "   OK   $FILE" || echo "   DIFF $FILE have=${A:-missing} pkg=$B"
  done
done
echo "dropbear: $(sha256sum /opt/sbin/dropbear | cut -c1-16) $(opkg status dropbear | awk '/^Version:/ {print $2}')"
echo "=== 4. MEDIUM ==="
dmesg 2>/dev/null | grep -iE 'i/o error|ext4|ext3|ext2|usb .*reset|sd[a-z].*error|illegal|segfault' | tail -n 8 | cut -c1-150
mount | grep ' /opt ' | cut -c1-100
} 2>&1 | tee -a "$OUT"
rm -rf $W
echo "SUMMARY: lines=$(wc -l < "$OUT")"
send_out
echo "RESULT SENT (OSC52)"
echo "RAW: $LAST"
EOF
(sh -n /tmp/dg12.sh && sh /tmp/dg12.sh)

=== OPTIONAL PASTE 3 (not read-only: a second SSH server for 3 minutes) ===
Starts Entware's dropbear in the foreground on port 2222 with its errors on the
screen, ends by itself after 180 s (or Ctrl+C). Port 22 and open sessions are not
touched. While it runs, connect from the phone: ssh -p 2222 root@<router>.
The lines it prints (and «Illegal instruction»/«Segmentation fault») are the answer.
K=""; for H in /opt/etc/dropbear/dropbear_*_host_key; do K="$K -r $H"; done
timeout 180 /opt/sbin/dropbear -F -E -p 2222 -P /tmp/dg12-db.pid $K 2>&1 | head -n 60
rm -f /tmp/dg12-db.pid
